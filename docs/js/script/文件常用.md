

> `file-type` 支持不全，使用 `magic` 补充

```js
const { fileTypeFromFile } = require('file-type')
const { Blake3Hasher } = require('@rasmx/blake3-node')
const { createReadStream } = require('fs')
const { basename, extname } = require('path')
const mmm = require('@vishalkumar14/mmmagic');

async function calculateBlake3(path) {
    const hasher = new Blake3Hasher();

    for await (const chunk of createReadStream(path)) {
        hasher.update(chunk);
    }

    return hasher.digest();
}

async function analyzeFile(filePath) {
    const hash = await calculateBlake3(filePath);
    let fileType = await fileTypeFromFile(filePath);
    if (!fileType) {
        const magic = new mmm.Magic(mmm.MAGIC_MIME_TYPE);
        fileType = {
            mime: await magic.detectFile(filePath),
            ext: extname(filePath).slice(1)
        }
    }
    return {
        hash,
        fileType,
    };
}

console.log(await analyzeFile("D:/Desktop/photo/a.py"))

// {
//     hash: 'd12ab9d2e1a7d067aca0e1c97fb0d36f271dce334d221acbb4acc72fd5e5fb46',
//     fileType: { mime: 'text/x-script.python', ext: 'py' }
// }
```
